# Format: <role> <tree path> -- <explanation>. "## <label>" starts a section, drawn as a tinted band.
# Roles: d directory; a app, l agent loop, m talks to the model (the colors of Pi's packages);
# w where you work; r read and run; o other.
d pi-2/ (python branch) -- pi/2 for Python
## the agent
a ├── pi2/ -- the agent, one file per part
a │   ├── cli.py -- options, loads extensions (-e)
a │   ├── tui.py -- interactive mode: editor, footer, tool boxes
a │   ├── session.py -- AgentSession: runs the loop, saves messages
l │   ├── agent_loop.py -- ask, run tool calls, repeat
a │   ├── model_runtime.py -- models.json, API key, stream_fn
m │   ├── ai.py -- OpenAI-compatible streaming provider
m │   ├── schema.py -- checks tool arguments
a │   ├── compaction.py -- summarizes old messages when the context fills up
a │   ├── windows.py -- Windows: stops a command's container and processes
a │   └── extensions.py -- define_tool, ExtensionAPI, loads -e files
## your work
w ├── exercises/ -- your extensions, each with a README and checks
w │   ├── 01-read/ … 04-edit/ -- you implement execute() in read.py … edit.py
w │   ├── 05-subagent/ -- a tool that runs a second agent
w │   └── check.py -- checks your tools without a model
w ├── playground/ -- for the agent: a failing test, 30 incident reports
## read and run
r ├── docs/ -- connecting a model; compaction; Windows
r ├── examples/ -- hello.py (a tool), permission_gate.py (a listener)
r ├── pi2.py -- starts pi2: python3 pi2.py -e ...
## outside the workshop
o ├── pyproject.toml -- pip install -e . gives a pi2 command
o └── assets/ -- logo, diagrams, the playground's generator
